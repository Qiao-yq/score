## 1. NeonSlider（霓虹发光评分滑块）

```tsx
import { useId } from 'react';
import { motion } from 'framer-motion';

interface NeonSliderProps {
  label: string;
  value: number;            // 0–100 整数
  onChange: (v: number) => void;
  accent?: 'cyan' | 'pink' | 'yellow';
  disabled?: boolean;
}

const ACCENT = {
  cyan:   { fg: '#00f3ff', glow: '0 0 8px rgba(0,243,255,.6)' },
  pink:   { fg: '#ff00aa', glow: '0 0 8px rgba(255,0,170,.6)' },
  yellow: { fg: '#fcee0a', glow: '0 0 8px rgba(252,238,10,.6)' },
} as const;

export function NeonSlider({ label, value, onChange, accent = 'cyan', disabled }: NeonSliderProps) {
  const id = useId();
  const c = ACCENT[accent];
  const pct = (value / 100) * 100;

  return (
    <div className="w-full">
      <div className="mb-1 flex items-center justify-between font-mono text-sm">
        <label htmlFor={id} className="text-hi tracking-wider">{label}</label>
        <span className="text-glow font-display text-lg" style={{ color: c.fg }}>
          {value}
        </span>
      </div>

      <input
        id={id}
        type="range"
        min={0}
        max={100}
        step={1}
        value={value}
        disabled={disabled}
        onChange={(e) => onChange(Number(e.target.value))}
        aria-label={label}
        aria-valuemin={0}
        aria-valuemax={100}
        aria-valuenow={value}
        className="neon-range w-full"
        style={{
          // 用背景渐变表现进度填充；thumb 发光
          background: `linear-gradient(90deg, ${c.fg} ${pct}%, #1a1633 ${pct}%)`,
          ['--thumb-glow' as any]: c.glow,
        }}
      />
    </div>
  );
}
```

配套 CSS：

```css

}

}

}
```

## 2. GlitchButton（故障效果按钮）

```tsx
import { motion } from 'framer-motion';
import type { ReactNode } from 'react';

interface GlitchButtonProps {
  children: ReactNode;
  onClick?: () => void;
  variant?: 'primary' | 'danger' | 'ghost';
  disabled?: boolean;
}

const VARIANT = {
  primary: 'text-void bg-cyan',
  danger:  'text-void bg-pink',
  ghost:  'text-cyan border border-cyan bg-transparent',
} as const;

export function GlitchButton({ children, onClick, variant = 'primary', disabled }: GlitchButtonProps) {
  return (
    <motion.button
      type="button"
      onClick={onClick}
      disabled={disabled}
      whileTap={{ scale: 0.97 }}
      className={`glitch-btn relative overflow-hidden px-5 py-2 font-display tracking-widest uppercase
                  clip-corner disabled:opacity-40 ${VARIANT[variant]}`}
      data-text={typeof children === 'string' ? children : undefined}
    >
      {/* 光扫过 */}
      <span aria-hidden className="scan-sweep" />
      {children}
    </motion.button>
  );
}
```

配套 CSS（Glitch 抖动 + 光扫）：

```css
.clip-corner { clip-path: polygon(0 0, calc(100% - 10px) 0, 100% 10px, 100% 100%, 10px 100%, 0 calc(100% - 10px)); }
.scan-sweep {
  position: absolute; inset: 0;
  background: linear-gradient(100deg, transparent 30%, rgba(255,255,255,.35) 50%, transparent 70%);
  transform: translateX(-120%);
  transition: transform .5s ease;
}
.glitch-btn:hover .scan-sweep { transform: translateX(120%); }
.glitch-btn:hover { animation: glitch-shift .3s steps(2) infinite; }
@keyframes glitch-shift {
  0%   { transform: translate(0); }
  25%  { transform: translate(-1px, 1px); }
  50%  { transform: translate(1px, -1px); }
  75%  { transform: translate(-1px, 0); }
  100% { transform: translate(0); }
}
@media (prefers-reduced-motion: reduce) {
  .glitch-btn:hover { animation: none; }
  .scan-sweep { transition: none; }
}
```

## 3. ConnectionStatus（连接状态呼吸灯）

```tsx
export type ConnState = 'synced' | 'reconnecting' | 'offline';

const STATE = {
  synced:      { color: '#00ff9f', label: '已同步', pulse: true },
  reconnecting:{ color: '#fcee0a', label: '重连中', pulse: true },
  offline:     { color: '#ff2d55', label: '连接中断', pulse: false },
} as const;

export function ConnectionStatus({ state }: { state: ConnState }) {
  const s = STATE[state];
  return (
    <div className="flex items-center gap-2 font-mono text-xs tracking-wider">
      <span
        className={`h-2.5 w-2.5 rounded-full ${s.pulse ? 'animate-pulse' : ''}`}
        style={{ background: s.color, boxShadow: `0 0 8px ${s.color}` }}
        aria-hidden
      />
      <span style={{ color: s.color }}>{s.label}</span>
    </div>
  );
}
```

## 4. DimensionScorePanel（教师复核维度面板）

```tsx
import { NeonSlider } from './NeonSlider';

interface DimensionState {
  key: string;
  label: string;
  weight: number;
  agentScore: number;
  confidence: number;
  teacherScore: number;
  action: 'approve' | 'suggest_modify' | 'insufficient';
  reason: string;
}

export function DimensionScorePanel({ dim }: { dim: DimensionState }) {
  const needsAdmin = Math.abs(dim.agentScore - dim.teacherScore) > 20
    || Math.abs(dim.teacherScore - dim.agentScore) / 100 > 0.20;

  return (
    <div className="glass clip-corner border border-cyan/40 p-4">
      <div className="mb-3 flex items-center justify-between">
        <div>
          <h3 className="font-display text-hi tracking-wider">{dim.label}</h3>
          <span className="font-mono text-xs text-dim">权重 {dim.weight * 100}%</span>
        </div>
        <span
          className={`font-mono text-xs ${dim.confidence < 0.70 ? 'text-pink' : 'text-success'}`}
        >
          置信度 {dim.confidence.toFixed(2)}
          {dim.confidence < 0.70 && ' · 需复核'}
        </span>
      </div>

      <div className="mb-2 font-mono text-sm text-base">
        Agent 分 <span className="text-cyan">{dim.agentScore}</span>
        <span className="mx-2 text-dim">→</span>
        教师分 <span className="text-yellow">{dim.teacherScore}</span>
      </div>

      <NeonSlider
        label="教师分"
        value={dim.teacherScore}
        onChange={() => {}}
        accent="yellow"
      />

      {needsAdmin && (
        <p className="mt-2 font-mono text-xs text-pink">⚠ 差异/改幅超阈值，需管理员二次确认</p>
      )}
    </div>
  );
}
```

## 6. 大屏 TOP3 排名卡片（缩略）

```tsx
import { motion } from 'framer-motion';

interface RankItem { rank: number; teamName: string; score: number; delta: number }

export function RankingPodium({ items }: { items: RankItem[] }) {
  return (
    <div className="flex items-end gap-6">
      {items.map((it, i) => (
        <motion.div
          key={it.teamName}
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ delay: i * 0.1 }}
          className="clip-corner border border-cyan/40 bg-deep/60 p-6 text-center backdrop-blur"
        >
          <div className="font-display text-6xl text-glow text-cyan">{it.rank}</div>
          <div className="mt-2 font-display text-xl text-hi">{it.teamName}</div>
          <div className="font-mono text-3xl text-yellow text-glow">{it.score.toFixed(1)}</div>
          <div className={`font-mono text-sm ${it.delta >= 0 ? 'text-success' : 'text-danger'}`}>
            {it.delta >= 0 ? '▲' : '▼'} {Math.abs(it.delta)}
          </div>
        </motion.div>
      ))}
    </div>
  );
}
```

## 7. 动效分级 Hook（核心/增强/可关闭）

```tsx
import { createContext, useContext, useMemo } from 'react';

export type MotionLevel = 'core' | 'enhanced' | 'reduced';

const MotionCtx = createContext<MotionLevel>('enhanced');

export function useMotionLevel(): MotionLevel {
  return useContext(MotionCtx);
}

export function resolveMotionLevel(prefersReduced: boolean, lowEnd: boolean): MotionLevel {
  if (prefersReduced) return 'reduced';
  if (lowEnd) return 'core';
  return 'enhanced';
}
```

> 组件按级别渲染：`core` 只保留反馈/过渡；`enhanced` 加 glitch/扫描线/代码雨；`reduced` 关闭非必要动效（粒子/3D/音效）。
